% =========================================================
% Shared LaTeX preamble for all papers/*/tex/paper.tex
% Single source of truth for styling/formatting.
% pdfTeX-only compatible (CI uses pdflatex/latexmk).
% =========================================================

\usepackage[letterpaper,margin=1in]{geometry}
\usepackage{iftex}

% ---------- Fonts / Unicode (pdfTeX-only in CI) ----------
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
  \usepackage{lmodern}
\else
  % We keep this branch inert for now to avoid requiring XeLaTeX/LuaLaTeX.
  % If we later enable xelatex/lualatex in CI, we can switch this on.
\fi

% ---------- Typography ----------
\usepackage{microtype}
\UseMicrotypeSet[protrusion]{basicmath}
\usepackage{setspace}
\setstretch{1.12}

% Whitepaper-style paragraphs
\setlength{\parindent}{0pt}
\setlength{\parskip}{6pt plus 1pt minus 1pt}
\raggedbottom

% ---------- Colors ----------
\usepackage{xcolor}
\definecolor{BrandAccent}{HTML}{204060}
\definecolor{BrandGray}{HTML}{6E6E6E}
\definecolor{RuleGray}{HTML}{D9D9D9}

% ---------- Headers / Footers ----------
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}

% Optional metadata hooks (defaults blank)
\newcommand{\docshorttitle}{}
\newcommand{\docversion}{}

\fancyfoot[L]{\small\textcolor{BrandGray}{\docshorttitle}}
\fancyfoot[C]{\small\textcolor{BrandGray}{\thepage}}
\fancyfoot[R]{\small\textcolor{BrandGray}{\docversion}}

% ---------- Hyperlinks ----------
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=BrandAccent,
  urlcolor=BrandAccent,
  citecolor=BrandAccent
}

% ---------- Boxes ----------
\usepackage[most]{tcolorbox}
\tcbset{
  enhanced,
  sharp corners,
  boxrule=0.8pt,
  colback=white,
  colframe=BrandAccent,
  left=10pt,
  right=10pt,
  top=8pt,
  bottom=8pt,
  before skip=8pt,
  after skip=8pt,
  breakable
}
\newtcolorbox{keytakeaway}{}

% ---------- Pandoc compatibility ----------
\providecommand{\tightlist}{}
\providecommand{\subtitle}[1]{}
\providecommand{\textsubscript}[1]{$_{\text{#1}}$}
\providecommand{\passthrough}[1]{#1}

% ---------- Executive Summary ----------
\newenvironment{execsummary}{\section*{Executive Summary}}{}
