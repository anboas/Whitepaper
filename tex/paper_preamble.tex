% Typography rules (repo-wide):
% - Use hyphen (-) for compounds; use en-dash (--) for numeric ranges (e.g., 6--12, 12--24 months).
% - Use em-dash (---) sparingly for parenthetical breaks in prose, with no surrounding spaces.
% - Use TeX quotes for prose: ``like this'' (avoid Unicode smart quotes and raw ASCII double-quotes in prose).
% - Keep code/config blocks (e.g., yamlblock) literal; do not typographically "prettify" their content.
% =========================================================

% PDF metadata (papers can define these macros)
\AtBeginDocument{%
  % NOTE: do not name temp macros starting with "\_" (that's the underscore command).
  \def\PDFMetaTitle{\ShortTitle}%
  \def\PDFMetaAuthor{}%
  \def\PDFMetaSubject{White Paper}%
  \def\PDFMetaKeywords{whitepaper, agentic autonomy, trust scopes, AI, defense}%
  \ifdefined\PaperTitle\def\PDFMetaTitle{\PaperTitle}\fi
  \ifdefined\AuthorName\def\PDFMetaAuthor{\AuthorName}\fi

  % 1) hyperref metadata (also feeds hyperxmp)
  \hypersetup{%
    pdftitle={\PDFMetaTitle},%
    pdfauthor={\PDFMetaAuthor},%
    pdfsubject={\PDFMetaSubject},%
    pdfkeywords={\PDFMetaKeywords}%
  }%

  % 2) pdfTeX info dictionary compatibility
  % We set ONLY /Author here.
  % hyperref already writes /Title and /Subject; duplicating keys triggers qpdf warnings and CI failures.
  \pdfstringdef\PDFAuthorString{\PDFMetaAuthor}%
  \pdfinfo{%
    /Author (\PDFAuthorString)
  }%
}
